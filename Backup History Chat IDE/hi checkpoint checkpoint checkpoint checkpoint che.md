แต่จากที่คุณถาม คุณต้องการให้รายการที่**ปล่อยไว้ไม่กดอะไร (pending without action)** ต้องแสดงว่า **"ยังไม่รับสินค้า"** ในไฟล์ Excel

## ✅ วิธีแก้ไข: ให้รายการที่ปล่อยไว้แสดงสถานะ "ยังไม่รับสินค้า" ใน Excel

ปัจจุบันระบบมี **2 สถานการณ์** ที่ต้องจัดการ:

### 1. **กรณีที่กด "ยังไม่รับสินค้า" (Skipped) เอง**
   - มี record ใน `MonthlyOrderActions` 
   - Action = "Skipped"
   - สถานะ: **"ยังไม่รับสินค้า"** ✅

### 2. **กรณีที่ปล่อยไว้ไม่กดอะไรเลย**
   - **ไม่มี** record ใน `MonthlyOrderActions` 
   - รายการอยู่ใน `pendingOrders` แต่ไม่ได้อยู่ใน `existingActions`
   - สถานะ: **ควรเป็น "ยังไม่รับสินค้า"** ก็ได้ เพราะมันยังไม่ได้รับจริงๆ

## 💡 Logic ที่ถูกต้อง

ตามหลักการบัญชีและการเงิน:

| สถานะ | ความหมาย | Export Excel ควรแสดง |
|---|---|---|
| **กดรับสินค้าแล้ว** | รับสินค้า + จ่ายเงินเต็มจำนวน | "รับสินค้าแล้ว" + ยอดเงิน |
| **กดผ่อนชำระ** | รับสินค้า + จ่ายเงินบางส่วน | "ผ่อนชำระ" + ยอดเงินงวดนี้ |
| **กดยังไม่รับสินค้า** | ของยังไม่มา / ยังไม่จ่าย | "ยังไม่รับสินค้า" + ยอดเงิน 0 บาท |
| **ปล่อยไว้ (ไม่กด)** | ยังไม่ตัดสินใจ = ยังไม่ได้รับของ | **"ยังไม่รับสินค้า"** + ยอดเงิน 0 บาท |

**เหตุผล:** ถ้ายังไม่กดอะไร แปลว่า **ยังไม่ได้ดำเนินการ = ยังไม่ได้รับของจริง** ดังนั้นควรแสดงว่า "ยังไม่รับสินค้า"

function processSection(ss, sheetName, groups, fallbackRows, headers, type) {
var sheet = ss.getSheetByName(sheetName);
if (!sheet) {
sheet = ss.insertSheet(sheetName);
} else {
sheet.clear();
}
var totalSaved = 0;
if (groups && groups.length > 0) {
for (var i = 0; i < groups.length; i++) {
var grp = groups[i];
if (grp.Rows && grp.Rows.length > 0) {
appendMonthBlock(sheet, grp.MonthYear, headers, grp.Rows, type);
totalSaved += grp.Rows.length;
}
}
}
else if (fallbackRows && fallbackRows.length > 0) {
appendMonthBlock(sheet, "ข้อมูลสำรอง", headers, fallbackRows, type);
totalSaved += fallbackRows.length;
}
applyColumnWidths(sheet, type);
return totalSaved;
}

if (type === "action") {

function formatThaiMonthTitle(key) {
if (!key) return "เดือน: ไม่ระบุ";
if (key.indexOf("-") !== -1) {
var parts = key.split("-");
var year = parseInt(parts[0], 10);
var month = parseInt(parts[1], 10);
if (!isNaN(year) && !isNaN(month)) {
if (year < 2500) year += 543;
var months = ["", "มกราคม", "กุมภาพันธ์", "มีนาคม", "เมษายน", "พฤษภาคม", "มิถุนายน",
"กรกฎาคม", "สิงหาคม", "กันยายน", "ตุลาคม", "พฤศจิกายน", "ธันวาคม"];
var mName = months[month] || ("เดือน " + month);
return "เดือน: " + mName + " " + year;
}
}
return "เดือน: " + key;
}
